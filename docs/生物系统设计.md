## 🎯 核心函数设计

### 1. 创建生物实例
```javascript
/**
 * 从模板创建生物实例
 * @param {string} templateId - 模板ID
 * @returns {Object} 生物实例
 */
function createCreature(templateId) {
  const template = CREATURE_TEMPLATES[templateId];
  const stats = THREAT_LEVELS[template.threatLevel];
  
  return {
    id: generateId(),
    templateId: templateId,
    name: template.name,
    
    ac: stats.ac,
    hp: {
      current: stats.hp,
      max: stats.hp
    },
    damage: stats.damage,
    attackBonus: stats.attackBonus,
    
    isAlive: true,
    conditions: [],
    
    abilities: template.abilities,
    weaknesses: template.weaknesses,
    loot: template.loot,
    behavior: template.behavior,
    
    description: template.description
  };
}
```

### 2. 生物攻击
```javascript
/**
 * 生物攻击
 * @param {Object} creature - 生物
 * @param {number} targetAC - 目标AC
 * @returns {Object} 攻击结果
 */
function creatureAttack(creature, targetAC) {
  // 命中检定
  const roll = getDice('d20').value;
  const total = roll + creature.attackBonus;
  
  const hit = total >= targetAC || roll === 20;
  const crit = roll === 20;
  const miss = roll === 1;
  
  let damage = 0;
  if (hit && !miss) {
    // 掷伤害
    const [diceCount, diceSize, bonus] = parseDamage(creature.damage);
    for (let i = 0; i < diceCount; i++) {
      damage += getDice(`d${diceSize}`).value;
    }
    damage += bonus || 0;
    
    if (crit) damage *= 2;
  }
  
  return {
    hit,
    crit,
    miss,
    roll,
    total,
    damage,
    creature: creature.name
  };
}
```

### 3. 受到伤害
```javascript
/**
 * 生物受到伤害
 * @param {Object} creature - 生物
 * @param {number} damage - 伤害值
 */
function damageCreature(creature, damage) {
  creature.hp.current = Math.max(0, creature.hp.current - damage);
  
  if (creature.hp.current === 0) {
    creature.isAlive = false;
    console.log(`${creature.name}被击败！`);
  }
}
```

### 4. 战利品掉落
```javascript
/**
 * 生物死亡掉落战利品
 * @param {Object} creature - 生物
 * @returns {Array} 战利品列表
 */
function dropLoot(creature) {
  const drops = [];
  
  creature.loot.forEach(lootEntry => {
    const roll = Math.random();
    if (roll < lootEntry.chance) {
      drops.push(lootEntry.item);
    }
  });
  
  // 经验值
  const template = CREATURE_TEMPLATES[creature.templateId];
  const stats = THREAT_LEVELS[template.threatLevel];
  const exp = Math.floor(
    stats.exp[0] + Math.random() * (stats.exp[1] - stats.exp[0])
  );
  
  return {
    items: drops,
    exp: exp,
    gold: Math.floor(exp / 5)  // 经验/5 = 金币
  };
}
```
